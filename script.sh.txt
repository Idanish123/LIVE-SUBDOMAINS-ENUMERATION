#!/bin/bash

# ===================================================================
# Advanced Subdomain Enumeration Framework (Refined Edition)
# Features: CLI Flags, IP Filtering, crt.sh, Pipefail, Format Preservation
# ===================================================================

# 1. SET -O PIPEFAIL: Ensures errors in a pipeline are captured
set -euo pipefail

# ------------------- DEFAULT CONFIGURATION -------------------------
INPUT_FILE=""
OUTPUT_FILE="recon_results.txt"
THREADS=50
TIMEOUT=10
SEPARATOR="#############################################################"
TEMP_DIR="/tmp/recon_$$"

# Tools toggle
USE_SUBFINDER=1
USE_ASSETFINDER=1
USE_FINDOMAIN=1
USE_CRTSH=1
USE_DNSX=1
USE_HTTPX=1

# ------------------- COLOR CODES -----------------------------------
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
CYAN='\033[0;36m'
NC='\033[0m'
BOLD='\033[1m'

# ------------------- FUNCTIONS -------------------------------------

show_help() {
    echo -e "${CYAN}Usage: $0 -i <input_file> [-o <output_file>]${NC}"
    echo -e ""
    echo -e "Options:"
    echo -e "  -i    Input file containing URLs or domains (required)"
    echo -e "  -o    Output file for results (default: recon_results.txt)"
    echo -e "  -h    Show this help message"
    echo -e ""
    echo -e "Example: $0 -i targets.txt -o final_report.txt"
}

check_dependencies() {
    local deps=("subfinder" "assetfinder" "findomain" "dnsx" "httpx" "anew" "jq" "curl")
    echo -e "${BLUE}[*] Validating toolkit...${NC}"
    for dep in "${deps[@]}"; do
        if ! command -v "$dep" &> /dev/null; then
            echo -e "${RED}[-] Error: $dep is not installed.${NC}"
            exit 1
        fi
    done
}

extract_domain() {
    echo "$1" | sed -E 's|^(https?://)?(www\.)?||' | sed -E 's|/.*||' | sed -E 's|:.*||' | tr '[:upper:]' '[:lower:]'
}

is_ip() {
    # Regex to detect IPv4 addresses to avoid wasting time
    if [[ $1 =~ ^([0-9]{1,3}\.){3}[0-9]{1,3}$ ]]; then
        return 0 # It is an IP
    else
        return 1 # It is not an IP
    fi
}

# ------------------- MAIN EXECUTION --------------------------------

# 2. CLI FLAG PARSING (-h / CLI flags)
while getopts "i:o:h" opt; do
    case "$opt" in
        i) INPUT_FILE=$OPTARG ;;
        o) OUTPUT_FILE=$OPTARG ;;
        h) show_help; exit 0 ;;
        *) show_help; exit 1 ;;
    esac
done

# Check if input file was provided
if [[ -z "$INPUT_FILE" ]]; then
    echo -e "${RED}[-] Error: Input file (-i) is required.${NC}"
    show_help
    exit 1
fi

if [[ ! -f "$INPUT_FILE" ]]; then
    echo -e "${RED}[-] Error: $INPUT_FILE not found!${NC}"
    exit 1
fi

check_dependencies
mkdir -p "$TEMP_DIR"
trap 'rm -rf "$TEMP_DIR"; exit' INT TERM EXIT

# Clear output file
> "$OUTPUT_FILE"

echo -e "${GREEN}${BOLD}Starting Reconnaissance...${NC}"

while IFS= read -r line || [ -n "$line" ]; do
    [[ -z "$line" || "$line" =~ ^[[:space:]]*# ]] && continue

    # 3. IP ADDRESS FILTERING
    if is_ip "$line"; then
        echo -e "${YELLOW}[!] Skipping IP Address: $line (DNS enumeration on IPs is a waste of time)${NC}"
        continue
    fi

    domain=$(extract_domain "$line")
    echo -e "\n${CYAN}[*] Targeting: ${BOLD}$domain${NC}"
    
    # Header in Output
    {
        echo "$SEPARATOR"
        echo "TARGET: $domain"
        echo "DATE: $(date)"
        echo "$SEPARATOR"
    } >> "$OUTPUT_FILE"

    # Create a temporary file for raw subdomains
    RAW_SUBS="$TEMP_DIR/raw_subs.txt"
    touch "$RAW_SUBS"

    # 4. CRT.SH INTEGRATION (Medium Severity Fix)
    if [ $USE_CRTSH -eq 1 ]; then
        echo -e "${BLUE}    └─ Querying crt.sh...${NC}"
        curl -s "https://crt.sh/?q=%$domain%&output=json" | jq -r '.[].name_value' 2>/dev/null | sed 's/\*\.//g' | sort -u >> "$RAW_SUBS" || true
    fi

    # Passive Tools
    if [ $USE_SUBFINDER -eq 1 ]; then
        subfinder -d "$domain" -silent >> "$RAW_SUBS" 2>/dev/null || true
    fi
    if [ $USE_ASSETFINDER -eq 1 ]; then
        assetfinder --subs-only "$domain" >> "$RAW_SUBS" 2>/dev/null || true
    fi
    if [ $USE_FINDOMAIN -eq 1 ]; then
        findomain -t "$domain" -q >> "$RAW_SUBS" 2>/dev/null || true
    fi

    # De-duplicate raw list
    sort -u "$RAW_SUBS" -o "$RAW_SUBS"

    # 5. DNS & HTTP PROBING (Fixing the "Loses > prefix" Critical Bug)
    # Instead of overwriting the output file, we probe the raw file 
    # and then format the output as we append it to the final file.
    
    PROBED_SUBS="$TEMP_DIR/probed.txt"
    touch "$PROBED_SUBS"

    if [ -s "$RAW_SUBS" ]; then
        echo -e "${BLUE}    └─ Resolving and Probing Live Hosts...${NC}"
        
        # DNSX to ensure they actually exist
        dnsx -l "$RAW_SUBS" -silent -a > "$TEMP_DIR/resolved.txt" || true
        
        # HTTPX to find live web services
        if [ -s "$TEMP_DIR/resolved.txt" ]; then
            httpx -l "$TEMP_DIR/resolved.txt" -silent -threads $THREADS -timeout $TIMEOUT > "$PROBED_SUBS" || true
        fi

        # Check if anything was found after probing
        if [ -s "$PROBED_SUBS" ]; then
            count=$(wc -l < "$PROBED_SUBS")
            echo -e "${GREEN}    └─ Found $count live subdomains.${NC}"
            
            # Format with prefix and append to output
            while IFS= read -r sub; do
                # We use 'awk' or 'sed' here to ensure we don't break the line
                echo -e "\t> $sub" >> "$OUTPUT_FILE"
            done < "$PROBED_SUBS"
        else
            echo -e "${RED}    └─ No live web services found.${NC}"
            echo "[-] No live services found" >> "$OUTPUT_FILE"
        fi
    else
        echo -e "${RED}    └─ No subdomains discovered.${NC}"
    fi

    echo -e "\n$SEPARATOR\n" >> "$OUTPUT_FILE"

done < "$INPUT_FILE"

echo -e "${GREEN}${BOLD}[+] Task Complete! Results: $OUTPUT_FILE${NC}"
